import { useState, useEffect } from 'react'
import { PatientTrackerService } from './generated/services/PatientTrackerService'
import type { PatientTrackerRead } from './generated/models/PatientTrackerModel'
import './App.css'

function statusColor(status?: string): string {
  switch (status?.toLowerCase()) {
    case 'admitted': return '#2563eb'
    case 'discharged': return '#16a34a'
    case 'critical': return '#dc2626'
    case 'observation': return '#d97706'
    default: return '#6b7280'
  }
}

function formatDate(dateStr?: string): string {
  if (!dateStr) return '—'
  return new Date(dateStr).toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' })
}

function PatientCard({ patient }: { patient: PatientTrackerRead }) {
  const status = patient.Visit_x0020_Type?.Value
  const doctor = patient.Doctor_x0020_Assigned?.DisplayName

  return (
    <div className="patient-card">
      <div className="card-header">
        <div className="patient-avatar">{patient.Title?.charAt(0) ?? '?'}</div>
        <div className="card-header-info">
          <h3 className="patient-name">{patient.Title ?? 'Unknown Patient'}</h3>
          <p className="patient-email">{patient.Email ?? '—'}</p>
        </div>
        {status && (
          <span className="status-badge" style={{ backgroundColor: statusColor(status) }}>
            {status}
          </span>
        )}
      </div>

      <div className="card-body">
        <div className="info-row">
          <span className="info-label">Date of Birth</span>
          <span className="info-value">{formatDate(patient.Date_x0020_of_x0020_Birth)}</span>
        </div>
        <div className="info-row">
          <span className="info-label">Gender</span>
          <span className="info-value">{patient.Gender?.Value ?? '—'}</span>
        </div>
        <div className="info-row">
          <span className="info-label">Assigned Doctor</span>
          <span className="info-value">{doctor ?? '—'}</span>
        </div>
        <div className="info-row">
          <span className="info-label">Primary Diagnosis</span>
          <span className="info-value">{patient.Notes ?? '—'}</span>
        </div>
        <div className="info-row">
          <span className="info-label">Admission Date</span>
          <span className="info-value">{formatDate(patient.Date)}</span>
        </div>
        <div className="info-row">
          <span className="info-label">Discharge Date</span>
          <span className="info-value">{formatDate(patient.Discharge_x0020_Date)}</span>
        </div>
        {patient.Notes && (
          <div className="clinical-notes">
            <span className="info-label">Clinical Notes</span>
            <p className="notes-text">{patient.Notes}</p>
          </div>
        )}
      </div>
    </div>
  )
}

function App() {
  const [patients, setPatients] = useState<PatientTrackerRead[]>([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    PatientTrackerService.getAll()
      .then((result) => {
        setPatients(result.data ?? [])
      })
      .catch((err) => {
        setError(err?.message ?? 'Failed to load patient data.')
      })
      .finally(() => {
        setLoading(false)
      })
  }, [])

  return (
    <div className="app-container">
      <header className="app-header">
        <h1 className="app-title">Patient Tracker</h1>
        <p className="app-subtitle">{!loading && !error ? `${patients.length} patient${patients.length !== 1 ? 's' : ''}` : ''}</p>
      </header>

      {loading && (
        <div className="state-message">
          <div className="spinner" />
          <p>Loading patients...</p>
        </div>
      )}

      {error && (
        <div className="state-message error">
          <p>{error}</p>
        </div>
      )}

      {!loading && !error && patients.length === 0 && (
        <div className="state-message">
          <p>No patients found.</p>
        </div>
      )}

      {!loading && !error && patients.length > 0 && (
        <div className="cards-grid">
          {patients.map((patient) => (
            <PatientCard key={patient.ID ?? patient['{Identifier}']} patient={patient} />
          ))}
        </div>
      )}
    </div>
  )
}

export default App
